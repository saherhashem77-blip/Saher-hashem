wrap.querySelector('#backup-restore-input').addEventListener('change', (e)=>{ const f = e.target.files[0]; if(!f) return; if(!confirm('سيتم استبدال كل البيانات والإعدادات الحالية بمحتوى الملف المختار. هل تريد الاستمرار؟')) return; const reader = new FileReader(); reader.onload = async (evt)=>{ try{ const data = JSON.parse(evt.target.result); if(data.config) saveConfig(data.config); if(Array.isArray(data.records)){ await dbClearAll(); for(const rec of data.records){ delete rec.id; await dbAdd(rec); } } toast('تمت استعادة البيانات بنجاح ✅'); setTimeout(()=> location.reload(), 1000); }catch(err){ toast('فشلت استعادة البيانات، الملف غير صالح ❌'); } }; reader.readAsText(f); }); } /* ---------------- بدء تشغيل التطبيق والاتصال بقاعدة البيانات ---------------- */ openDB().then(()=>{ renderApp(); }).catch(err=>{ console.error('فشل فتح قاعدة البيانات المحلية:', err); renderApp(); });
